    internal? func classifyScore(score: R|Int64|): R|String| {
        RETURN {
            MATCH {
                BRANCH(expr(score < 60)) {
                    STRING("fail")
                }
                BRANCH(expr(score < 70)) {
                    STRING("pass")
                }
                BRANCH(expr(score < 90)) {
                    STRING("good")
                }
                BRANCH(_) {
                    STRING("excellent")
                }
            }
        }
    }
    internal? func extractYear(time: R|TimeUnit|): R|Int32| {
        RETURN {
            MATCH {
                subject:
                    NAMED_ACCESS(time)
                BRANCH(Year(deferred(y)) where 1 == 1) {
                    NAMED_ACCESS(y)
                }
                BRANCH(Year(deferred(y)) | Month(deferred(y), _)) {
                    NAMED_ACCESS(y)
                }
            }
        }
    }
