    internal? func controlFlowTest(x: R|Int64|): R|Int64| {
        IF {
            condition:
                COMPARISON(GT) {
                    QUALIFIED_ACCESS(x)
                    INT(0)
                }
            then:
                RETURN {
                    QUALIFIED_ACCESS(x)
                }
            else:
                RETURN {
                    FUNCTION_CALL(unaryMinus) {
                        receiver:
                            QUALIFIED_ACCESS(x)
                    }
                }
        }
    }
