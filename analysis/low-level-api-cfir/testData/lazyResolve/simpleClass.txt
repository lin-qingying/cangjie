    package sample.lowlevel.lazy
    [ResolvedTo(BODY_RESOLVE)] internal? class Box<T> {
        [ResolvedTo(BODY_RESOLVE)] public abstract? prop value: R|T|
        [ResolvedTo(BODY_RESOLVE)] init([ResolvedTo(BODY_RESOLVE)] value: R|T|) {
            ASSIGNMENT {
                lValue:
                    NAMED_ACCESS(value -> CfirPropertySymbol(value)) {
                        receiver:
                            <element: CfirThisReceiverExpression>
                    }
                rValue:
                    NAMED_ACCESS(value -> CfirValueParameterSymbol(value))
            }
        }
        [ResolvedTo(BODY_RESOLVE)] public func unwrap(): R|T| {
            RETURN {
                NAMED_ACCESS(value -> CfirPropertySymbol(value))
            }
        }
    }
    [ResolvedTo(BODY_RESOLVE)] internal? func consume([ResolvedTo(BODY_RESOLVE)] box: R|Box<Int64>|): R|Int64| {
        RETURN {
            FUNCTION_CALL(unwrap -> CfirNamedFunctionSymbol(unwrap)) {
                receiver:
                    NAMED_ACCESS(box -> CfirValueParameterSymbol(box))
            }
        }
    }
